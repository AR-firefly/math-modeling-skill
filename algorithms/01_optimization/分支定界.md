# 分支定界

## 1 适用情况

- 数据要求：整数规划（max c'x，约束 A x <= b，变量非负整数），规模中到小（变量/约束几十个内）
- 场景：需要整数变量精确解、不想引入 MILP 求解器的轻量场景，或教学演示分支剪枝过程
- 何时用：整数变量个数不多、想输出"分支树"与剪枝依据
- 何时不用：大规模整数规划（优先 milp）、只需要近似解（用元启发式）

## 2 可联动算法

- 与线性规划连续松弛对比（LP 上界 vs 整数解），量化"整数约束的代价"
- 大规模时降级为遗传算法/模拟退火求近似，用分支定界验证小规模子问题
- 与 `线性整数0-1规划.md` 的 milp 互为校验（milp 提供标准答案）

## 3 完整代码模板

```python
"""分支定界求整数规划精确解（最大化）。"""
import numpy as np


def branch_and_bound(c, A, b):
    """最大化 c'x，约束 A x <= b，x>=0 整数。返回 (最优值, 最优解)。"""
    from scipy.optimize import linprog
    c = np.asarray(c, float)
    A = np.asarray(A, float)
    b = np.asarray(b, float)
    n = len(c)
    best_val, best_x = -np.inf, None

    def lp_relax(lb, ub):
        """解连续松弛，返回 (目标函数值, 解)。"""
        res = linprog(-c, A_ub=A, b_ub=b,
                      bounds=list(zip(lb, ub)), method="highs")
        return (None, None) if not res.success else (res.fun, res.x)

    def dfs(lb, ub):
        nonlocal best_val, best_x
        f, x = lp_relax(lb, ub)
        if x is None or -f <= best_val + 1e-9:      # 剪枝：不可行或不如当前界
            return
        frac = [i for i in range(n) if abs(x[i] - round(x[i])) > 1e-6]
        if not frac:                                # 全整数，更新上界
            best_val, best_x = -f, x.copy()
            return
        i = frac[0]                                 # 对首个小数变量分支
        lo, hi = lb.copy(), ub.copy()
        hi[i] = np.floor(x[i])                      # 左支：x_i <= floor
        dfs(lo, hi)
        lo2, hi2 = lb.copy(), ub.copy()
        lo2[i] = np.floor(x[i]) + 1                 # 右支：x_i >= floor+1
        dfs(lo2, hi2)

    dfs(np.zeros(n), np.full(n, np.inf))
    return best_val, best_x


if __name__ == "__main__":
    # max 8x1 + 11x2 + 6x3，5x1 + 7x2 + 4x3 <= 14，x>=0 整数，最优值 22
    c = np.array([8.0, 11.0, 6.0])
    A = np.array([[5.0, 7.0, 4.0]])
    b = np.array([14.0])
    val, x = branch_and_bound(c, A, b)
    assert abs(val - 22.0) < 1e-6                  # 已知最优值 22
    assert A @ x <= b + 1e-9 and np.all(x >= -1e-9)  # 可行
    assert np.allclose(x, np.round(x))             # 整数解
    print("最优值:", val, "最优解:", x)
```
