# 动态规划

## 1 适用情况

- 数据要求：问题有"最优子结构 + 重叠子问题"，状态转移可表达（背包、资源分配、多阶段决策、最短路）
- 场景：背包问题、生产-库存计划、投资分配、序列对齐
- 何时用：决策分阶段、每阶段状态有限、能用递推表求解，规模不大（状态数千万级内）
- 何时不用：状态爆炸（维度高时换整数规划/启发式）、无重叠子问题（直接用递归/枚举）

## 2 可联动算法

- 与 0-1 整数规划（milp）对比，DP 更轻、无需求解器
- 与贪心对比：DP 保证最优、贪心只保证近似，量化差距
- 多阶段生产/库存可联动模拟退火做大规模近似

## 3 完整代码模板

```python
"""0-1 背包动态规划：给定重量 w、价值 v、容量 W，求最大价值。"""
import itertools
import numpy as np


def knapsack_dp(w, v, W):
    """返回 (最大价值, 选中的物品下标列表)。"""
    w = list(w)
    v = list(v)
    n = len(w)
    dp = np.zeros((n + 1, W + 1), dtype=int)
    for i in range(1, n + 1):
        for cap in range(W + 1):
            if w[i - 1] <= cap:                                  # 装或不装取大
                dp[i, cap] = max(dp[i - 1, cap],
                                 dp[i - 1, cap - w[i - 1]] + v[i - 1])
            else:
                dp[i, cap] = dp[i - 1, cap]
    # 回溯被选物品
    cap, pick = W, []
    for i in range(n, 0, -1):
        if dp[i, cap] != dp[i - 1, cap]:
            pick.append(i - 1)
            cap -= w[i - 1]
    return int(dp[n, W]), pick[::-1]


if __name__ == "__main__":
    # demo：随机 8 件物品，容量 20，与全枚举最优值对比
    rng = np.random.RandomState(42)
    w = rng.randint(1, 10, 8)
    v = rng.randint(10, 50, 8)
    W = 20
    val, pick = knapsack_dp(w, v, W)
    # 全枚举校验（2^8 = 256 种组合）
    best = 0
    for r in range(len(w) + 1):
        for idx in itertools.combinations(range(len(w)), r):
            if sum(w[i] for i in idx) <= W:
                best = max(best, sum(v[i] for i in idx))
    assert val == best                              # 与枚举最优一致
    assert sum(w[i] for i in pick) <= W             # 总重量不超容量
    print("重量:", w, "价值:", v)
    print("DP 最大价值:", val, "选中:", pick, "枚举最优:", best)
```
