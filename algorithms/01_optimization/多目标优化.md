# 多目标优化

## 1 适用情况

- 数据要求：多个冲突目标函数 fs（均最小化），给定变量边界 bounds
- 场景：成本-效率、精度-时间、多个冲突指标的权衡决策
- 何时用：目标互相制约、需要输出 Pareto 前沿供决策者挑选折衷解
- 何时不用：目标可加权合成单一目标（直接单目标优化）、不关心前沿分布

## 2 可联动算法

- ε-约束法：一个目标进约束、另一目标求极值，扫出 Pareto 前沿
- 与加权和对比：加权和只能得到一个点，ε-约束能出完整前沿
- 前沿结果可联动 TOPSIS 选"最理想折衷解"

## 3 完整代码模板

```python
"""多目标优化：ε-约束法求双目标 Pareto 前沿。"""
import numpy as np


def epsilon_constraint(fs, bounds, n=50):
    """fs: [f1, f2] 均最小化；bounds: 各变量 (min, max) 列表。
    返回 Pareto 前沿点列表 [(f1, f2), ...]（已去重去支配）。"""
    from scipy.optimize import minimize
    x0 = np.array([(a + b) / 2 for a, b in bounds])
    # 求 f2 的极值区间 [f2_at_f1min, f2max]
    r1 = minimize(lambda x: fs[0](x), x0, method="SLSQP", bounds=bounds)
    r2 = minimize(lambda x: fs[1](x), x0, method="SLSQP", bounds=bounds)
    lo_f2 = fs[1](r1.x)
    hi_f2 = fs[2](r2.x) if len(fs) > 2 else fs[1](r2.x)
    raw = []
    for i in range(n):
        eps = lo_f2 + (hi_f2 - lo_f2) * i / (n - 1)
        con = {"type": "ineq", "fun": lambda x, e=eps: e - fs[1](x)}  # f2 <= eps
        res = minimize(lambda x: fs[0](x), x0, method="SLSQP",
                       bounds=bounds, constraints=[con])
        if res.success:
            raw.append((fs[0](res.x), fs[1](res.x)))
    # 去重 + 保留非支配点（f1 升序下 f2 严格递减 → Pareto 前沿）
    raw = sorted(set((round(a, 4), round(b, 4)) for a, b in raw))
    front, minf2 = [], np.inf
    for p in raw:
        if p[1] < minf2 - 1e-9:
            front.append(p)
            minf2 = p[1]
    return front


if __name__ == "__main__":
    # demo：ZDT1 简化版（2 变量）min f1 = x1，min f2 = g·(1-sqrt(x1/g))，g = 1+9x2
    def f1(x):
        return float(x[0])

    def f2(x):
        g = 1 + 9 * x[1]
        return float(g * (1 - np.sqrt(x[0] / g)))

    front = epsilon_constraint([f1, f2], [(0, 1), (0, 1)], n=50)
    assert len(front) > 1                       # 前沿有多个折衷点
    assert all(0 <= a <= 1 and 0 <= b <= 1 for a, b in front)   # 目标范围合理
    # 前沿随 f1 增大 f2 应非增
    assert all(front[i + 1][1] <= front[i][1] + 1e-6 for i in range(len(front) - 1))
    print("Pareto 前沿点数:", len(front), "示例:", front[:5], "...")
```
