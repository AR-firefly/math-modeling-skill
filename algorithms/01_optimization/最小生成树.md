# 最小生成树

## 1 适用情况

- 数据要求：带权图（邻接矩阵 adj 或边列表 (u, v, w)），求连通全部节点的最小总权
- 场景：网络布线、通信/输电线路铺设、聚类（单链接）
- 何时用：目标是把所有点连起来且总成本最小、无向连通图
- 何时不用：要求两点间最短路（用 Dijkstra/Floyd）、图有方向性（MST 概念失效）

## 2 可联动算法

- Prim 与 Kruskal 互验（同一图两算法总权一致）
- MST 可作聚类依据（移除最长边得到簇）
- 作为网络设计的下界，与 Steiner 树等更复杂方案对比

## 3 完整代码模板

```python
"""最小生成树：Prim（邻接矩阵）与 Kruskal（边列表）双实现。"""
import numpy as np


def mst_prim(adj):
    """Prim：adj 邻接矩阵，返回 (总权值, 边列表)。"""
    n = adj.shape[0]
    in_mst = np.zeros(n, dtype=bool)
    dist = np.full(n, np.inf)
    parent = np.full(n, -1)
    dist[0] = 0
    total, edges = 0.0, []
    for _ in range(n):
        u = int(np.argmin(np.where(in_mst, np.inf, dist)))   # 选未入树最近点
        in_mst[u] = True
        if parent[u] != -1:
            total += dist[u]
            edges.append((int(parent[u]), u))
        for v in range(n):                                   # 更新邻接点
            if not in_mst[v] and adj[u, v] < dist[v]:
                dist[v], parent[v] = adj[u, v], u
    return total, edges


def mst_kruskal(edges):
    """Kruskal：edges 为 (u, v, w) 列表，返回 (总权值, 边列表)。"""
    parent = {}

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    total, mst_edges = 0.0, []
    for u, v, w in sorted(edges, key=lambda e: e[2]):        # 按权升序
        parent.setdefault(u, u)
        parent.setdefault(v, v)
        ru, rv = find(u), find(v)
        if ru != rv:                                         # 不成环才加入
            parent[ru] = rv
            total += w
            mst_edges.append((u, v))
    return total, mst_edges


if __name__ == "__main__":
    # demo：6 节点随机无向图，两算法结果一致
    rng = np.random.RandomState(42)
    n = 6
    w = rng.uniform(0.5, 5.0, (n, n))
    adj = (w + w.T) / 2
    np.fill_diagonal(adj, 0)
    edges = [(i, j, adj[i, j]) for i in range(n) for j in range(i + 1, n)]
    t1, e1 = mst_prim(adj)
    t2, e2 = mst_kruskal(edges)
    assert abs(t1 - t2) < 1e-9               # Prim 与 Kruskal 总权一致
    assert len(e1) == n - 1 and len(e2) == n - 1   # 树有 n-1 条边
    print("Prim 总权:", round(t1, 4), "| Kruskal 总权:", round(t2, 4))
```
