# 禁忌搜索

## 1 适用情况

- 数据要求：cost 函数（最小化）接收"排列/指派"类解，定义好邻域（交换/插入）
- 场景：二次分配（QAP）、旅行商、排程、布局——组合优化逃出局部最优
- 何时用：邻域搜索会陷局部最优、希望短期记忆禁止回跳
- 何时不用：连续优化（用 PSO/DE）、规模很小（枚举即可）

## 2 可联动算法

- 用贪心/随机生成初始解，加快起步
- 与模拟退火、蚁群算法对比组合优化质量
- 结尾加 2-opt 局部搜索精修 TSP 类结果

## 3 完整代码模板

```python
"""禁忌搜索（Tabu Search）：排列型解的邻域搜索，最小化 cost。"""
import itertools
import numpy as np


def tabu_search(cost, init, tabu_len=10, iters=100):
    """init: 初始排列（下标数组）；邻域 = 交换两个位置。
    返回 (最优解, 最优代价)。"""
    cur = np.asarray(init, dtype=int).copy()
    best, best_cost = cur.copy(), cost(cur)
    cur_cost = best_cost
    tabu = []                                     # 禁忌表：禁止回退的交换对
    n = len(cur)
    pairs = list(itertools.combinations(range(n), 2))   # 全部交换对
    for _ in range(iters):
        best_nei, best_nei_cost, best_pair = None, np.inf, None
        for (i, j) in pairs:
            if (i, j) in tabu or (j, i) in tabu:  # 禁忌的交换跳过
                continue
            nxt = cur.copy()
            nxt[i], nxt[j] = nxt[j], nxt[i]
            c = cost(nxt)
            if c < best_nei_cost:
                best_nei_cost, best_nei, best_pair = c, nxt, (i, j)
        if best_nei is None:
            break
        cur = best_nei
        cur_cost = best_nei_cost
        tabu.append(best_pair)                    # 记录禁忌并滑动窗口
        if len(tabu) > tabu_len:
            tabu.pop(0)
        if cur_cost < best_cost:
            best, best_cost = cur.copy(), cur_cost
    return best, best_cost


if __name__ == "__main__":
    # demo：5 个设施的二次分配（QAP），cost = Σ D[i,j] * F[p[i],p[j]]
    rng = np.random.RandomState(42)
    D = rng.uniform(0, 1, (5, 5)); D = (D + D.T) / 2      # 距离矩阵（对称）
    F = rng.uniform(0, 1, (5, 5)); F = (F + F.T) / 2      # 流量矩阵（对称）
    np.fill_diagonal(D, 0); np.fill_diagonal(F, 0)
    init = np.arange(5)
    rng.shuffle(init)

    def qap_cost(p):
        p = np.asarray(p)
        return float((D * F[np.ix_(p, p)]).sum())

    best, val = tabu_search(qap_cost, init, tabu_len=10, iters=200)
    # 全枚举校验最优（5! = 120 种排列）
    brute = min(qap_cost(p) for p in itertools.permutations(range(5)))
    assert val <= qap_cost(init)                  # 不劣于初始解
    assert abs(val - brute) < 1e-6                # 达到全局最优
    print("禁忌搜索最优代价:", round(val, 4), "最优排列:", best)
```
