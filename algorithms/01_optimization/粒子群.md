# 粒子群

## 1 适用情况

- 数据要求：fitness 函数（最小化）接收连续向量，可带边界 bounds
- 场景：连续参数优化、多峰连续寻优、参数标定
- 何时用：变量连续、目标多峰、无梯度信息，需群体搜索全局解
- 何时不用：光滑凸目标（用 scipy 梯度法更快）、变量离散（用 GA/整数规划）

## 2 可联动算法

- 与差分进化、模拟退火对比求解质量，多跑几次报均值±方差
- 结果作初始点喂给局部优化（L-BFGS）精修
- 灵敏度分析：扫描粒子数/惯性权重观察收敛性

## 3 完整代码模板

```python
"""粒子群优化（PSO）：最小化连续函数。"""
import numpy as np


def pso(fitness, dim=2, bounds=(-5, 5), n=30, iters=100, seed=42):
    """fitness: 接收 (dim,) 向量；bounds: 每维上下界（标量或元组对）。
    返回 (最优位置, 最优值)。"""
    rng = np.random.RandomState(seed)
    lo, hi = bounds[0], bounds[1]
    x = rng.uniform(lo, hi, (n, dim))                    # 位置
    v = rng.uniform(-(hi - lo), hi - lo, (n, dim)) * 0.2  # 速度
    pbest = x.copy()
    gbest = x[np.argmin([fitness(p) for p in x])].copy()  # 全局最优
    w, c1, c2 = 0.7, 1.5, 1.5                              # 惯性 + 自我 + 社会
    for _ in range(iters):
        for i in range(n):
            if fitness(x[i]) < fitness(pbest[i]):
                pbest[i] = x[i].copy()
        gbest = pbest[np.argmin([fitness(p) for p in pbest])].copy()
        r1, r2 = rng.rand(n, dim), rng.rand(n, dim)
        v = w * v + c1 * r1 * (pbest - x) + c2 * r2 * (gbest - x)
        x = np.clip(x + v, lo, hi)                        # 更新并限界
    k = int(np.argmin([fitness(p) for p in x]))
    return x[k], float(fitness(x[k]))


if __name__ == "__main__":
    # demo：Rastrigin f = 2*10 + Σ(x_i^2 - 10cos(2πx_i))，全局最小 0 在原点
    def rastrigin(x):
        return float(2 * 10 + np.sum(x ** 2 - 10 * np.cos(2 * np.pi * x)))

    x, val = pso(rastrigin, dim=2, bounds=(-5, 5), n=30, iters=100, seed=42)
    assert val < 1.0                     # 收敛到较小的值（接近全局最优 0）
    assert np.linalg.norm(x) < 3.0       # 最优位置靠近原点
    print("最优位置:", x, "最优值:", val)
```
