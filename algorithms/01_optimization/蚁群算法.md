# 蚁群算法

## 1 适用情况

- 数据要求：距离/代价矩阵（对称或非对称），路径类组合优化问题
- 场景：旅行商（TSP）、车辆路径、配送、测线/巡检路径
- 何时用：组合搜索空间大、需要借"信息素+启发式"平衡探索与利用
- 何时不用：规模极小（枚举更快）、连续优化（换 PSO/DE）

## 2 可联动算法

- 用贪心最近邻求上界作对照，验证 ACO 提升
- 结果可接局部搜索（2-opt）精修
- 与禁忌搜索、遗传算法对比路径质量

## 3 完整代码模板

```python
"""蚁群算法（ACO）解对称 TSP：dist 为距离矩阵。"""
import numpy as np


def aco(dist, n_ants=10, iters=50, alpha=1, beta=2, rho=0.5, seed=42):
    """返回 (最短路径长度, 路径含起点终点)。"""
    rng = np.random.RandomState(seed)
    dist = np.asarray(dist, float)
    n = dist.shape[0]
    tau = np.ones((n, n))                        # 信息素矩阵
    best_len, best_path = np.inf, None
    for _ in range(iters):
        for _ in range(n_ants):
            start = int(rng.randint(n))
            path = [start]
            unvisited = set(range(n)); unvisited.remove(start)
            cur = start
            while unvisited:                     # 依概率选择下一城市
                nxts = list(unvisited)
                eta = 1.0 / (dist[cur, nxts] + 1e-12)     # 启发式 = 1/距离
                p = tau[cur, nxts] ** alpha * eta ** beta
                p = p / p.sum()
                cur = nxts[int(rng.choice(len(nxts), p=p))]
                path.append(cur)
                unvisited.remove(cur)
            length = (sum(dist[path[i], path[i + 1]] for i in range(n - 1))
                      + dist[path[-1], start])               # 走完再回起点
            if length < best_len:
                best_len, best_path = length, path + [start]
        # 信息素蒸发 + 增强最优路径
        tau *= (1 - rho)
        for i in range(n):
            tau[best_path[i], best_path[i + 1]] += 1.0 / best_len
    return best_len, best_path


if __name__ == "__main__":
    # demo：8 个随机点 TSP（seed=42 固定，可复现）
    rng = np.random.RandomState(42)
    pts = rng.uniform(0, 10, (8, 2))
    D = np.sqrt(((pts[:, None, :] - pts[None, :, :]) ** 2).sum(-1))

    # 贪心最近邻上界（参考基线）
    def nearest_upper():
        unvisited = set(range(8)); cur = 0; unvisited.remove(0)
        total = 0.0
        while unvisited:
            nxt = min(unvisited, key=lambda j: D[cur, j])
            total += D[cur, nxt]; unvisited.remove(nxt); cur = nxt
        return total

    length, path = aco(D, n_ants=20, iters=100, seed=42)  # 适度增强保证收敛
    assert len(set(path)) == 8                    # 遍历全部城市
    assert path[0] == path[-1]                     # 闭合回路（首尾同城）
    assert abs(length - sum(D[path[i], path[i + 1]] for i in range(8))) < 1e-9
    assert length <= 2.0 * nearest_upper() + 1e-9  # 合理性：启发式不产生荒谬路径
    print("ACO 路径长度:", round(length, 3), "路径:", path)
```
