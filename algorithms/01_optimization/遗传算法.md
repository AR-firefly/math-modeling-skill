# 遗传算法

## 1 适用情况

- 数据要求：fitness 函数（最大化）接收编码后的解（如 0/1 位串），不要求可导
- 场景：组合爆炸优化、布局、调度、特征选择、参数编码优化
- 何时用：解空间离散/巨大、搜索无结构、需要并行群体搜索
- 何时不用：连续光滑目标且规模小（用 scipy 梯度/差分进化更快更稳）

## 2 可联动算法

- 精英保留 + 局部搜索（爬山）形成 GA+局部搜索混合，专门对付组合爆炸（2025A）
- 用贪心/最近邻生成部分初始种群，加速收敛
- 与模拟退火、粒子群对比，多跑几次报均值±方差

## 3 完整代码模板

```python
"""遗传算法（GA）：0/1 位串编码，锦标赛选择 + 单点交叉 + 位变异，最大化 fitness。"""
import numpy as np


def ga(fitness, n_bits, pop=50, gens=200, pc=0.8, pm=0.1, seed=42):
    """fitness: 接收 0/1 位串数组；n_bits: 编码长度。返回 (最优位串, 最优适应度)。"""
    rng = np.random.RandomState(seed)
    p = rng.randint(0, 2, (pop, n_bits))             # 初始种群
    for _ in range(gens):
        fits = np.array([fitness(x) for x in p])
        elite = p[int(np.argmax(fits))].copy()          # 精英保留
        # 锦标赛选择（两两挑优，加列维扩展以广播到每个位）
        i, j = rng.randint(pop, size=pop), rng.randint(pop, size=pop)
        sel = np.where((fits[i] > fits[j])[:, None], p[i], p[j])
        # 单点交叉
        nxt = []
        for k in range(0, pop, 2):
            a, b = sel[k].copy(), sel[k + 1].copy()
            if rng.rand() < pc:
                pt = rng.randint(1, n_bits)
                a[pt:], b[pt:] = b[pt:].copy(), a[pt:].copy()
            nxt += [a, b]
        nxt = np.array(nxt)
        # 位变异
        nxt ^= (rng.rand(pop, n_bits) < pm).astype(int)
        nxt[0] = elite                                 # 精英直接进入下一代
        p = nxt
    fits = np.array([fitness(x) for x in p])
    k = int(np.argmax(fits))
    return p[k], float(fits[k])


if __name__ == "__main__":
    # demo：one-max，最大化位串中 1 的个数，20 位应收敛到全 1
    rng = np.random.RandomState(42)
    n_bits = 20
    best, val = ga(lambda x: int(x.sum()), n_bits, pop=50, gens=200, pc=0.8, pm=0.1, seed=42)
    assert val == n_bits                # 找到全 1 位串
    assert best.sum() == n_bits
    print("最优适应度:", val, "位串:", best)
```
