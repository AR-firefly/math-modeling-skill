# 三角测量定位

## 1 适用情况

- 数据要求：若干已知点坐标 + 从每点观测到目标的方位角
- 场景：由测向数据（雷达/声呐/光学测角）反推目标位置
- 何时用：已知点 ≥ 2、观测方向线近似共点交会
- 何时不用：测的是距离而非角度（改用三边定位/距离最小二乘）

## 2 可联动算法

- 坐标变换：把各测点本地坐标系旋到全局再交会
- 几何建模：交会本质是线线相交几何
- 最小二乘拟合：角度含噪时加权最小二乘提升鲁棒性

## 3 完整代码模板

```python
"""三角测量定位：由已知点+观测方位角，最小二乘求视线交点。"""
import numpy as np


def triangulate(points, angles):
    """points: (k,2) 已知点坐标；angles: 对应方位角(弧度)，指向目标。
    视线方程 sinθ·(X-x) - cosθ·(Y-y) = 0，最小二乘解即目标位置。"""
    points = np.asarray(points, float)
    angles = np.asarray(angles, float)
    A, b = [], []
    for (x, y), th in zip(points, angles):
        A.append([np.sin(th), -np.cos(th)])
        b.append(np.sin(th) * x - np.cos(th) * y)
    X, *_ = np.linalg.lstsq(np.array(A), np.array(b), rcond=None)
    return X


if __name__ == "__main__":
    # demo：两个已知点 (0,0),(5,0) 观测目标 (3,4)
    target = np.array([3.0, 4.0])
    pts = np.array([[0.0, 0.0], [5.0, 0.0]])
    ang = np.arctan2(target[1] - pts[:, 1], target[0] - pts[:, 0])   # 测站指向目标的方位角
    pos = triangulate(pts, ang)
    assert np.linalg.norm(pos - target) < 1e-3, f"定位误差 {np.linalg.norm(pos-target)}"
    # 三点交会也收敛
    pts3 = np.array([[0.0, 0.0], [5.0, 0.0], [0.0, 5.0]])
    ang3 = np.arctan2(target[1] - pts3[:, 1], target[0] - pts3[:, 0])
    pos3 = triangulate(pts3, ang3)
    assert np.linalg.norm(pos3 - target) < 1e-3
    print(f"两点定位={pos}，三点定位={pos3}，目标={target}")
```
