# 光学干涉太阳几何

## 1 适用情况

- 数据要求：太阳几何——纬度、日期、时刻；光学干涉——缝距、波长、屏距
- 场景：太阳高度角/方位角计算、双缝/薄膜干涉明暗条纹定位
- 何时用：需日照/光伏倾角优化、光谱仪/干涉装置条纹计算
- 何时不用：需精确日出日落时刻（引入大气折射修正公式）

## 2 可联动算法

- FFT 周期提取：干涉条纹/光谱做频谱提取，读主频对应波长周期（联动 2025B）
- 运动学动力学：太阳高度角随时刻的运动轨迹建模
- 数值积分：条纹强度分布积分求能量

## 3 完整代码模板

```python
"""太阳高度角（近似公式）与双缝干涉亮纹间距。"""
import numpy as np


def sun_elevation(lat, lon, day, hour):
    """太阳高度角（度）。lat/lon: 经纬度(度)；day: 年积日 1-365；
    hour: 真太阳时(小时)。Cooper 近似，忽略时区与大气折射。"""
    phi = np.radians(lat)
    delta = np.radians(23.45 * np.sin(np.radians(360 * (day - 81) / 365)))  # 赤纬
    H = np.radians(15 * (hour - 12))            # 时角，正午为 0
    a = np.sin(phi) * np.sin(delta) + np.cos(phi) * np.cos(delta) * np.cos(H)
    return np.degrees(np.arcsin(np.clip(a, -1, 1)))


def interference(d, wavelength=500e-9, L=1.0):
    """双缝干涉相邻亮纹间距(m)：Δy = λL/d。d: 缝距(m)；L: 屏距(m)。"""
    return wavelength * L / d


if __name__ == "__main__":
    # demo1：纬度30° 春分日(day=80, 赤纬≈0) 正午高度角 ≈ 60°
    el = sun_elevation(30, 0, 80, 12)
    assert 0 <= el <= 90
    assert abs(el - (90 - 30)) < 1.0, f"正午高度角 {el} 应≈60°"
    # demo2：缝距 1mm、屏距 1m、绿光 500nm → 条纹间距 0.5mm
    dy = interference(1e-3, wavelength=500e-9, L=1.0)
    assert abs(dy - 5e-4) < 1e-8, f"条纹间距 {dy} 应=5e-4"
    print(f"正午高度角={el:.2f}°（预期≈60°），干涉条纹间距={dy*1e3:.4f}mm")
```
