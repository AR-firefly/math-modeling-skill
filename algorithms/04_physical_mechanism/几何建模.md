# 几何建模

## 1 适用情况

- 数据要求：点/线/面/三角形的坐标参数
- 场景：点线距离、点面距离、三角形面积等几何量计算
- 何时用：题目给出几何构型需量化计算、网格/面元化后度量几何量
- 何时不用：复杂曲面/体网格（用有限元或专业几何库）

## 2 可联动算法

- 坐标变换：先统一坐标系再算几何量
- 三角测量定位：视线交点本质是线线几何
- 运动学动力学：运动轨迹的几何包络/距离度量

## 3 完整代码模板

```python
"""几何建模：点到直线/平面距离、三角形面积（海伦公式）。"""
import numpy as np


def distance_geom(p, a, b, c=None):
    """p: 目标点；a,b: 直线上的两点；若给 c，则 a,b,c 确定平面。
    c=None 返回点到直线距离，否则返回点到平面距离。"""
    p, a, b = (np.asarray(v, float) for v in (p, a, b))
    if c is None:
        return np.linalg.norm(np.cross(p - a, b - a)) / np.linalg.norm(b - a)
    c = np.asarray(c, float)
    n = np.cross(b - a, c - a)          # 平面法向量
    n = n / np.linalg.norm(n)
    return abs(np.dot(p - a, n))        # 点到平面距离


def triangle_area(a, b, c):
    """海伦公式计算三角形面积。"""
    def L(u, v):
        return np.linalg.norm(np.asarray(u, float) - np.asarray(v, float))
    x, y, z = L(b, c), L(a, c), L(a, b)
    s = (x + y + z) / 2
    return np.sqrt(max(s * (s - x) * (s - y) * (s - z), 0.0))


if __name__ == "__main__":
    # demo1：点(1,1,1) 到 x 轴（过原点沿 x 方向）的距离 = √2
    d = distance_geom([1, 1, 1], [0, 0, 0], [1, 0, 0])
    assert abs(d - np.sqrt(2)) < 1e-10
    # demo2：3-4-5 直角三角形面积 = 6
    S = triangle_area([0, 0, 0], [3, 0, 0], [0, 4, 0])
    assert abs(S - 6) < 1e-10
    print(f"点线距离={d:.6f}（√2={np.sqrt(2):.6f}），三角形面积={S}")
```
