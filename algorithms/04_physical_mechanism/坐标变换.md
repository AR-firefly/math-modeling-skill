# 坐标变换

## 1 适用情况

- 数据要求：点/向量坐标、欧拉角或旋转轴角参数
- 场景：刚体姿态描述、坐标系换算、3D 位姿追踪（舞龙/系泊/机械臂）
- 何时用：需要旋转/平移到统一坐标系、验证正交不变性
- 何时不用：仅需平移缩放（用线性代数直接做）

## 2 可联动算法

- 几何建模：变换后算点线面距离/面积
- 三角测量定位：把传感器坐标系旋到全局坐标系再求交
- 运动学动力学：姿态旋转 + 运动方程联合建模

## 3 完整代码模板

```python
"""坐标变换：2D 旋转 / 3D 欧拉角旋转（角度制）。"""
import numpy as np


def rot_2d(p, theta):
    """二维旋转：p 逆时针旋转 theta 度。"""
    th = np.radians(theta)
    R = np.array([[np.cos(th), -np.sin(th)],
                  [np.sin(th), np.cos(th)]])
    return R @ np.asarray(p, float)


def rot_transform(p, angles):
    """三维欧拉角旋转 Z-Y-X'（角度制）。返回 (旋转后向量, 旋转矩阵 R)。"""
    ax, ay, az = np.radians(angles)
    Rx = np.array([[1, 0, 0],
                   [0, np.cos(ax), -np.sin(ax)],
                   [0, np.sin(ax), np.cos(ax)]])
    Ry = np.array([[np.cos(ay), 0, np.sin(ay)],
                   [0, 1, 0],
                   [-np.sin(ay), 0, np.cos(ay)]])
    Rz = np.array([[np.cos(az), -np.sin(az), 0],
                   [np.sin(az), np.cos(az), 0],
                   [0, 0, 1]])
    R = Rz @ Ry @ Rx
    return R @ np.asarray(p, float), R


if __name__ == "__main__":
    # demo：3D 旋转后长度不变（旋转矩阵正交）
    p = np.array([1.0, 2.0, 3.0])
    p2, R = rot_transform(p, [30, 45, 60])
    assert abs(np.linalg.norm(p2) - np.linalg.norm(p)) < 1e-10   # 长度保持
    assert np.allclose(R @ R.T, np.eye(3), atol=1e-10)           # R 正交
    # 2D：90° 把 (1,0) 旋到 (0,1)
    q = rot_2d(np.array([1.0, 0.0]), 90)
    assert np.allclose(q, [0, 1], atol=1e-10)
    print(f"3D旋转后={p2}，长度不变；2D 90° 旋转={q}")
```
